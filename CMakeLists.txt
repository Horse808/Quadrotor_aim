cmake_minimum_required(VERSION 3.10)
project(QuadrotorAimLightTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SDK_ROOT "/home/mafei/LicheeRV_Nano_Build/LicheeRV-Nano-Build-20260114" CACHE PATH "LicheeRV SDK")
set(OPENCV_DIR "" CACHE PATH "Nano OpenCV config")
set(BOARD_LIBS "${CMAKE_SOURCE_DIR}/build-nano/lib" CACHE PATH "Nano runtime libs")
if(NOT OPENCV_DIR)
 message(FATAL_ERROR "Set OPENCV_DIR to target OpenCV package")
endif()
set(OpenCV_DIR "${OPENCV_DIR}")
find_package(OpenCV REQUIRED COMPONENTS core imgproc geometry objdetect video)
execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -dumpmachine OUTPUT_VARIABLE TRIPLE OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT TRIPLE MATCHES "riscv64")
 message(FATAL_ERROR "Use RISC-V cross compiler, got ${TRIPLE}")
endif()
set(MW "${SDK_ROOT}/middleware/v2")
include_directories(src "${MW}/3rdparty/stb" ${OpenCV_INCLUDE_DIRS} "${MW}/include" "${MW}/include/isp/sg200x" "${MW}/modules/isp/include/sg200x" "${MW}/modules/sys/include" "${MW}/sample/common" "${SDK_ROOT}/osdrv/interdrv/v2/include/common/uapi" "${SDK_ROOT}/osdrv/interdrv/v2/include/chip/mars/uapi")
link_directories("${MW}/modules/isp/lib/musl_riscv64" "${BOARD_LIBS}" "${MW}/sample/test_mmf")
set(NANO_LIBS sample vpu sys venc vdec misc sns_full ini isp isp_algo ae af awb cvi_bin_isp cvi_bin)
set(PLATFORM_LIBS)
foreach(lib IN LISTS NANO_LIBS)
 if(NOT EXISTS "${BOARD_LIBS}/lib${lib}.so")
  message(FATAL_ERROR "Missing ${BOARD_LIBS}/lib${lib}.so")
 endif()
 list(APPEND PLATFORM_LIBS "${BOARD_LIBS}/lib${lib}.so")
endforeach()
if(EXISTS "${BOARD_LIBS}/libsns_gc4653.so")
 list(APPEND PLATFORM_LIBS "${BOARD_LIBS}/libsns_gc4653.so")
endif()
if(EXISTS "${BOARD_LIBS}/libjson-c.so")
 list(APPEND PLATFORM_LIBS "${BOARD_LIBS}/libjson-c.so")
elseif(EXISTS "${BOARD_LIBS}/libjson-c.so.5.1.0")
 list(APPEND PLATFORM_LIBS "${BOARD_LIBS}/libjson-c.so.5.1.0")
else()
 message(FATAL_ERROR "Missing target libjson-c.so")
endif()
add_executable(main src/main.cpp src/camera.cpp src/light_detector.cpp src/communicate.cpp src/state_machine.cpp)
add_executable(aim_light_test test/light_detection_preview.cpp src/camera.cpp src/light_detector.cpp)
add_executable(camera_compressed_stream_test test/camera_compressed_stream.cpp src/camera.cpp)
add_executable(camera_calibration_stream camera_calibration/camera_calibration_stream.cpp src/camera.cpp)
set_target_properties(main PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
set_target_properties(aim_light_test PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
set_target_properties(camera_compressed_stream_test PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
set_target_properties(camera_calibration_stream PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
target_link_libraries(main ${OpenCV_LIBS} ${PLATFORM_LIBS} pthread m dl atomic)
target_link_libraries(aim_light_test ${OpenCV_LIBS} ${PLATFORM_LIBS} pthread m dl atomic)
target_link_libraries(camera_compressed_stream_test ${OpenCV_LIBS} ${PLATFORM_LIBS} pthread m dl atomic)
target_link_libraries(camera_calibration_stream ${OpenCV_LIBS} ${PLATFORM_LIBS} pthread m dl atomic)

add_executable(main_test test/main_test.cpp test/ArUco_detector.cpp src/camera.cpp src/communicate.cpp src/state_machine.cpp)
set_target_properties(main_test PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
target_link_libraries(main_test ${OpenCV_LIBS} ${PLATFORM_LIBS} pthread m dl atomic)
